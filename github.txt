# 기존 Git 저장소가 있다면

cd C:\work\ncri
git remote -v
git add .
git commit -m "강의 목록/수강신청 및 후기 한글화 업데이트"
git push origin main


# 새 저장소를 만드는 경우

cd C:\work\ncri
git init
git branch -M main
git add .
git commit -m "초기 커밋"


# GitHub에서 빈 저장소 생성 후, 표시된 원격 주소 사용

git remote add origin https://github.com/ELiHong/ncri.git
git push -u origin main
